cmake_minimum_required(VERSION 3.16)
project(orangepi_hardware_sync LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MVS_ROOT "" CACHE PATH "Existing private ARM64 MVS SDK payload")
find_path(MVS_INCLUDE MvCameraControl.h PATHS "${MVS_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(MVS_LIBRARY MvCameraControl PATHS "${MVS_ROOT}/lib/aarch64" NO_DEFAULT_PATH REQUIRED)
get_filename_component(MVS_LIBRARY_DIR "${MVS_LIBRARY}" DIRECTORY)
add_executable(camera_probe camera_probe.cpp)
target_include_directories(camera_probe PRIVATE ${MVS_INCLUDE})
target_link_libraries(camera_probe PRIVATE ${MVS_LIBRARY})
set_target_properties(camera_probe PROPERTIES BUILD_RPATH "${MVS_LIBRARY_DIR}" INSTALL_RPATH "${MVS_LIBRARY_DIR}")
option(BUILD_ROS_CAPTURE "Build isolated ROS camera capture" ON)
if(BUILD_ROS_CAPTURE)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(cv_bridge REQUIRED)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
  add_executable(camera_sync camera_sync.cpp)
  target_compile_options(camera_sync PRIVATE -include cmath)
  install(TARGETS camera_sync DESTINATION bin)
  target_include_directories(camera_sync PRIVATE ${MVS_INCLUDE} ${OpenCV_INCLUDE_DIRS})
  ament_target_dependencies(camera_sync rclcpp sensor_msgs std_msgs cv_bridge)
  target_link_libraries(camera_sync ${MVS_LIBRARY} ${OpenCV_LIBS})
  set_target_properties(camera_sync PROPERTIES BUILD_RPATH "${MVS_LIBRARY_DIR}" INSTALL_RPATH "${MVS_LIBRARY_DIR}")
endif()
